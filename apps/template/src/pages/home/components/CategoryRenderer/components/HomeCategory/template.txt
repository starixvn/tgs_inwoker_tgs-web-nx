# 計算呈現LobbyCategory || NonLobbyCategory 的數量

  const showCategoryAmount = (tag: string, isLobbyType: boolean) => {
    if (isMobileSize) {
      if (isLobbyType) {
        return 3;
      } else {
        if (tag === "Hot") {
          return 9;
        }
      }
      return 6;
    }
    if (isDesktopSize) {
      return 6;
    }
  };

# HomeCategory 首頁渲染用

  {Object.values(lobbyCategoryList).map((category) => {
    const { key, iconUrl, text, gameType } = category;

    if (
      [
        GameTypeEnum.SLOT,
        GameTypeEnum.FISH,
        GameTypeEnum.TABLEGAME,
      ].includes(gameType)
    ) {
      //  slots, fishing, and egame (with games)
      return (
        <LobbyTagCategory
          key={key}
          iconUrl={iconUrl}
          text={text}
          gameType={gameType}
          gameTag={text}
          showDataLength={showCategoryAmount(text, false)!}
          handleViewAllClick={() => {
            handleGameTypeClick(gameType);
          }}
          gameTypeName={
            gameTypeList.find((gameTypeItem) => gameTypeItem.gameType === gameType)
              ?.key || ""
          }
          groupBy_gameType={Object.values(groupBy_gameType)}
        />
      );
    } else {
      //  others
      return (
        <LobbyCategory
          key={key}
          iconUrl={iconUrl}
          text={text}
          gameType={gameType}
          showDataLength={showCategoryAmount(text, true)!}
          handleViewAllClick={() => {
            handleGameTypeClick(gameType);
          }}
          gameTypeName={
            gameTypeList.find((gameTypeItem) => gameTypeItem.gameType === gameType)
              ?.key || ""
          }
        />
      );
    }
  })}
